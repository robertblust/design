# One modal — implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the family's three dialogs with one modal (`modal.js` and `modal.css`) in the terminal palette. It holds the graph viewer or the diagram viewer, closes with `Close · Esc`, and keeps the page behind it still. The chat's close says `Close · Esc` too.

**Architecture:**

- **One modal file.** `assets/modal.js` defines `window.rbModal`. It draws a native `<dialog class="rbmodal">` with a head (title, a controls slot, ×) and a body. It moves the nodes it is given in, and moves them back on close when they had a place.
- **Keyed dialogs.** A modal opened with a `key` is kept and reused, so a body that must never move (the graph's iframe, which a move would reload) stays inside it for the page's life.
- **Loading.** `chat.js` and `stage.js` fetch `modal.js` from beside themselves on the first open, and `modal.js` links `modal.css` from beside itself.
- **Palette.** The terminal palette moves into the tokens block, so the chat and the modal read one set of `--t-*` values.

**Tech Stack:** Plain browser JavaScript, CSS, `node --test`, Playwright Chromium.

**Spec:** `docs/superpowers/specs/2026-09-28-one-modal-design.md`. Read it first. The owner's reference image is `~/Desktop/modal.png`.

**Interfaces (produces):**

- `window.rbModal.open({ key?, title, body, controls?, opener?, onClose? }) → handle`
  - `body` is a node or an array of nodes.
  - A body node that has a parent is moved in, with a comment marker left where it stood, and moved back in front of that marker on close.
  - A body node without a parent stays in the dialog after close. That only makes sense with `key`.
  - `key` reuses the dialog for that key: its body is already inside, and `open` only updates the title and shows it.
- `handle.title(text)`, `handle.close()`, and `handle.el`, the dialog.
- `window.rbModal.labels({ close })`, which relabels every modal's close.
- The dialog has `aria-labelledby` pointing at its title.
- The close is `button.rbmodal-close` with `aria-label` and `data-tip` set to the close sentence.
- `modal.js` reads the close sentence from `window.rbModalWords` when a caller has set it. Otherwise it uses `{ en: "Close · Esc", de: "Schliessen · Esc" }`, by `<html lang>`, and relabels on a `lang` change through a MutationObserver.

```js
(function(){
  if (window.rbModal) return;
  var here = document.currentScript && document.currentScript.src;
  if (here && !document.querySelector('link[data-rbmodal]')) {
    var css = document.createElement("link"); css.rel = "stylesheet"; css.href = new URL("modal.css", here).href; css.setAttribute("data-rbmodal", "");
    document.head.appendChild(css);
  }
  var WORDS = { en: "Close · Esc", de: "Schliessen · Esc" };
  function closeWord(){ var w = window.rbModalWords || WORDS; return document.documentElement.lang === "de" ? w.de : w.en; }
  var open = 0, keyed = {}, all = [], saved = null;
  function lock(){ if (open++ === 0) { saved = { y: window.scrollY, o: document.documentElement.style.overflow }; document.documentElement.style.overflow = "hidden"; } }
  function unlock(){ if (--open === 0 && saved) { document.documentElement.style.overflow = saved.o; window.scrollTo(0, saved.y); saved = null; } }
  function make(){
    var d = document.createElement("dialog"); d.className = "rbmodal";
    var head = document.createElement("div"); head.className = "rbmodal-head";
    var title = document.createElement("span"); title.className = "rbmodal-title"; title.id = "rbmodal-title-" + all.length;
    var controls = document.createElement("div"); controls.className = "rbmodal-controls";
    var x = document.createElement("button"); x.type = "button"; x.className = "rbmodal-close"; x.textContent = "×";
    var body = document.createElement("div"); body.className = "rbmodal-body";
    head.appendChild(title); head.appendChild(controls); head.appendChild(x);
    d.appendChild(head); d.appendChild(body); d.setAttribute("aria-labelledby", title.id);
    x.addEventListener("click", function(){ d.close(); });
    d.addEventListener("click", function(ev){ if (ev.target === d) d.close(); });
    document.body.appendChild(d); all.push(d); label(d);
    return { el: d, titleEl: title, controls: controls, body: body, marks: [], onClose: null, opener: null, open: false };
  }
  function label(d){ var x = d.querySelector(".rbmodal-close"), w = closeWord(); x.setAttribute("aria-label", w); x.setAttribute("data-tip", w); }
  function relabel(){ all.forEach(label); }
  if (window.MutationObserver) new MutationObserver(relabel).observe(document.documentElement, { attributes: true, attributeFilter: ["lang"] });
  function openModal(o){
    var m = o.key && keyed[o.key] ? keyed[o.key] : make();
    if (o.key) keyed[o.key] = m;
    m.titleEl.textContent = o.title || "";
    if (o.controls && o.controls.parentNode !== m.controls) { m.controls.textContent = ""; m.controls.appendChild(o.controls); }
    [].concat(o.body || []).forEach(function(n){
      if (!n || n.parentNode === m.body) return;
      if (n.parentNode) { var mark = document.createComment("rbmodal"); n.parentNode.insertBefore(mark, n); m.marks.push([n, mark]); }
      m.body.appendChild(n);
    });
    m.onClose = o.onClose || null; m.opener = o.opener || document.activeElement;
    if (!m.bound) {
      m.bound = true;
      m.el.addEventListener("close", function(){
        m.marks.forEach(function(p){ if (p[1].parentNode) { p[1].parentNode.insertBefore(p[0], p[1]); p[1].parentNode.removeChild(p[1]); } });
        m.marks = []; m.open = false; unlock();
        var cb = m.onClose, back = m.opener; m.onClose = null; m.opener = null;
        if (cb) cb(); if (back && back.focus) back.focus({ preventScroll: true });
      });
    }
    if (!m.open) { m.open = true; lock(); m.el.showModal(); }
    return { el: m.el, title: function(t){ m.titleEl.textContent = t; }, close: function(){ if (m.el.open) m.el.close(); } };
  }
  window.rbModal = { open: openModal, labels: relabel };
})();
```
